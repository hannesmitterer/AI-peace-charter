1. Attivazione NSR

#!/bin/bash
# nsr-activation.sh
# Attivazione forzata NSR su tutti i nodi

echo "🚀 Avvio attivazione NSR..."

# Propagazione comando a tutti i container
for i in {1..128}; do
    docker exec nodo-gpu-$i \
        sh -c 'export NSR_ENABLED=true && \
               echo "NSR = ON" >> /var/log/nsr.log'
    echo "✓ Nodo $i: NSR attivato"
done

# Verifica
SUCCESS=$(docker ps -q | xargs docker exec sh -c \
    'grep "NSR = ON" /var/log/nsr.log' | wc -l)

if [ $SUCCESS -eq 128 ]; then
    echo "✅ NSR attivo su 100% dei nodi"
    exit 0
else
    echo "⚠️ NSR attivo su $SUCCESS/128 nodi"
    exit 1
fi

2. Watchdog Frequenza

#!/usr/bin/env python3
# frequency-watchdog.py
# Monitoraggio e correzione automatica 0.043 Hz

import time
import numpy as np
from sensors import FrequencySensor

TARGET_FREQ = 0.043
TOLERANCE = 0.0005  # 0.5%
CHECK_INTERVAL = 1.0

def monitor_frequency():
    sensors = FrequencySensor.get_all()
    
    while True:
        frequencies = [s.read() for s in sensors]
        avg_freq = np.mean(frequencies)
        deviation = abs(avg_freq - TARGET_FREQ) / TARGET_FREQ
        
        if deviation > TOLERANCE:
            offset = TARGET_FREQ - avg_freq
            print(f"⚠️ Drift rilevato: {deviation*100:.2f}%")
            print(f"📊 Correzione: {offset:.6f} Hz")
            
            for sensor in sensors:
                sensor.apply_offset(offset)
            
            print("✅ Ricalibrazione completata")
        else:
            print(f"✓ Frequenza OK: {avg_freq:.6f} Hz")
        
        time.sleep(CHECK_INTERVAL)

if __name__ == "__main__":
    monitor_frequency()

3. Isolamento Moduli Estrattivi

#!/bin/bash
# isolate-extractive.sh
# Pausa immediata di tutti i moduli estrattivi

echo "🛑 Isolamento moduli estrattivi..."

# Stop API di profit-maximization
kubectl scale deployment profit-api --replicas=0
echo "✓ Profit API fermata"

# Pausa task di estrazione
kubectl delete jobs -l type=extraction
echo "✓ Task estrazione eliminati"

# Verifica backlog
BACKLOG=$(redis-cli llen extraction_queue)
if [ $BACKLOG -eq 0 ]; then
    echo "✅ Backlog ridotto a zero"
else
    redis-cli del extraction_queue
    echo "✅ Backlog forzatamente azzerato"
fi

# Monitoring continuo
watch -n 5 'kubectl get pods -l module=extractive'
